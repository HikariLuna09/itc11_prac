import java.util.Scanner;

class Main {
    public static void main(String[] args) {
        int num1, num2, ans;
        Scanner plus = new Scanner (System.in);

        System.out.print("Enter num1: ");
        num1 = plus.nextInt();
        System.out.print("Enter num2: ");
        num2 = plus.nextInt();
        ans=(num1+num2)*2;

        System.out.println("Result: "+ans);
    }
}
